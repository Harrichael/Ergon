# Into another database, where the .ergon files still find their projects.
$ ergon export > backup.json
$ ERGON_DB=restored.db ergon import backup.json
Imported alpha (5 tasks)
Imported beta (1 task)
$ ERGON_DB=restored.db ergon tree
root  in_progress  p-  Alpha
  qxpj  todo         p1  Backend
    FTFE  in_progress  p1  API
  nHcx  todo         p-  Frontend  blocked
$ ERGON_DB=restored.db ergon project ls --all
alpha  2025-10-09 08:53 (just now)
beta  2025-10-09 08:53 (just now)  retired
$ ERGON_DB=restored.db ergon restore pQyk
Restored alpha/pQyk
