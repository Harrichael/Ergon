# The outline: a project's tasks as editable text, one per line, applied
# back in one transaction.

$ cd proj
$ ergon init demo "Build a demo" --no-launch
Created project demo in $ROOT/proj
$ ergon add "Backend\n\nDetails about the backend." --priority p1
qxpj
$ ergon add "Write the API" --parent qxpj
FTFE
$ ergon add "Document the API" --parent FTFE
nHcx
$ ergon add Frontend
pQyk
$ ergon dep add pQyk FTFE
$ ergon edit FTFE --status in_progress

# Slug, status box (none: follows the parent), first line of the intent,
# then the stored priority and dependencies.
$ ergon outline
# ergon demo 73b15015
qxpj [ ] Backend !p1
  FTFE [~] Write the API
    nHcx Document the API
pQyk [ ] Frontend deps:FTFE

# For editors: the same text, plus what inference makes of each line.
$ ergon outline --json

# Removing a line deletes the task with its subtasks, except the ones moved
# elsewhere. Dependents are counted like `ergon show` counts them.
$ ergon outline --apply <<EOF
# ergon demo e11f5837
FTFE [~] Write the API
  nHcx Document the API !p3
pQyk [ ] Frontend deps:dFR6,9cDQ
EOF
FTFE Write the API: move to the top level
pQyk Frontend: deps +9cDQ +dFR6
qxpj Backend services: delete, with its 1 subtask and 1 dependent
dFR6 Docs: delete, with 1 dependent
$ ergon outline

# `ergon edit` does, unless the outline changes their status too.
