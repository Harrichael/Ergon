# ergon gantt: one bar per open top-level task, placed by dependencies at
# their real depth. Work runs as soon as what it waits on is done, with no
# limit on how much runs at once, except that a sized task's steps run one
# after another; the axis counts size.

$ ergon init demo Root --no-launch
Created project demo in $ROOT

# API's two sized parts run side by side, so it spans 5, not 8. UI's
# Screens waits on API's Schema alone, and UI's Copy waits on nothing, so
# UI starts at 0, then waits (the dots) until Schema is done at 3.
$ ergon add API
qxpj
$ ergon add Schema --parent qxpj --size 3
FTFE
$ ergon add Handlers --parent qxpj --size 5
nHcx
$ ergon add UI
pQyk
$ ergon add Copy --parent pQyk --size 1
9cDQ
$ ergon add Screens --parent pQyk --size 8
dFR6
$ ergon dep add dFR6 FTFE
$ COLUMNS=80 ergon gantt
                      0   2   4   6   8   10
qxpj  todo         8  ██████████              API
pQyk  todo         9  ██····████████████████  UI
· waiting on other work

# Unsized work is drawn as 1. Docs waits on Review, so far the only step of
# Handlers, which takes all of its size: Docs starts at 5.
$ ergon add Docs
qwpT
$ ergon add Review --parent nHcx
AENA
$ ergon dep add qwpT AENA
$ COLUMNS=80 ergon gantt
                      0   2   4   6   8   10
qxpj  todo         8  ██████████              API
pQyk  todo         9  ██····████████████████  UI
qwpT  todo         -            ▒▒            Docs
▒ unsized, drawn as 1   · waiting on other work

# Landing, done and cancelled tasks don't show and hold nothing back: with
# Schema landing, Screens starts at once.
$ ergon edit FTFE --status landing
$ ergon add Shipped --status landing
FubR

▒ unsized, drawn as 1

# Work can sit between two steps of a sized task. Tests waits on Docs, which
# waits on Review: Review takes the first half of Handlers, Docs runs after
# it, and Tests takes the second half after Docs. Handlers waits between.
$ ergon add Tests --parent nHcx
Hwgf
$ ergon dep add Hwgf qwpT
$ COLUMNS=80 ergon gantt
                        0  1  2  3  4  5  6  7  8

pQyk  todo         9    ████████████████████████  UI
qwpT  todo         -           ▒▒▒                Docs
▒ unsized, drawn as 1   · waiting on other work

# Given a sized task, the bars are its steps, each a third of its size now.
# Review waits on Lint, so Lint goes ahead of it, though it's later in the
