# Sizes. A sized task is one unit of work; its subtasks are steps of it.
# Tasks above units add them up; views show sizes once any task has one.

$ ergon init demo Root --no-launch
Created project demo in $ROOT

$ ergon add Epic
qxpj
$ ergon add API --parent qxpj --size 5
FTFE
$ ergon add Routes --parent FTFE
nHcx
$ ergon add UI --parent qxpj --size 8
pQyk
$ ergon add Docs --parent qxpj
9cDQ
$ ergon tree
root  in_progress  p-  13+  Root
  qxpj  todo         p-  13+  Epic
    FTFE  todo         p-  5    API
      nHcx  todo         p-  -    Routes
    pQyk  todo         p-  8    UI
    9cDQ  todo         p-  -    Docs

# Sizes are Fibonacci numbers.
$ ergon edit pQyk --size 4
error: invalid value '4' for '--size <SIZE>': invalid size "4": expected a Fibonacci number: 1, 2, 3, 5, 8, 13, ...

For more information, try '--help'.
[exit 2]

# Sized tasks never nest, from either end.
$ ergon edit nHcx --size 2
error: refused: demo/FTFE and its subtask demo/nHcx can't both be sized: a size marks one unit of work, and its subtasks are steps of it. Unsize one: `ergon edit demo/FTFE --size none` or `ergon edit demo/nHcx --size none`
[exit 1]
$ ergon edit qxpj --size 13
error: refused: demo/qxpj and its subtask demo/FTFE can't both be sized: a size marks one unit of work, and its subtasks are steps of it. Unsize one: `ergon edit demo/qxpj --size none` or `ergon edit demo/FTFE --size none`
[exit 1]

# Done work leaves the total but not what's left; cancelled work leaves both.
$ ergon edit FTFE --status done
$ ergon edit 9cDQ --status cancelled
$ ergon ls
