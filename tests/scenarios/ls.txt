# ergon ls lists the coarsest open, unblocked work, most urgent first.

$ ergon init demo Root --no-launch
Created project demo in $ROOT
$ ergon add Parent --priority p2
qxpj
$ ergon add First --parent qxpj
FTFE
$ ergon add Second --parent qxpj
nHcx
$ ergon add Prerequisite
pQyk
$ ergon ls
qxpj  p2  todo         Parent
pQyk  p-  todo         Prerequisite

# Second blocked: Parent splits into its unblocked children. Prerequisite is
# boosted to Second's p2; equal priorities list oldest first.
$ ergon dep add nHcx pQyk
$ ergon ls
FTFE  p2  todo         First  (in: Parent)
pQyk  p2  todo         Prerequisite

# tree --ls marks the same tasks inside the tree.
$ ergon tree --ls
  root  in_progress  p-  Root
    qxpj  todo         p2  Parent
▸     FTFE  todo         p2  First
      nHcx  todo         p2  Second  blocked
▸   pQyk  todo         p2  Prerequisite
$ ergon tree qxpj --ls
  qxpj  todo         p2  Parent
▸   FTFE  todo         p2  First
    nHcx  todo         p2  Second  blocked

# Before Second was blocked, Parent was listed whole; --subtasks shows what's
# under each row.
$ ergon dep rm nHcx pQyk
$ ergon ls --subtasks

  nHcx  p2  todo         Second
pQyk  p-  todo         Prerequisite

# --sort orders by another field, ascending unless followed by desc. Tasks
# without a value go last either way; ties stay most urgent, then oldest.
$ cd sorting
$ ergon init sorting Root --no-launch
Created project sorting in $ROOT/sorting
$ ergon add Small --size 2 --attention 7 --priority p2
qxpj
$ ergon add Big --size 8 --attention 2 --priority p1
FTFE
$ ergon add Unrated --priority p0
nHcx
$ ergon ls
nHcx  p0  todo         -  a?   Unrated
FTFE  p1  todo         8  a2   Big
qxpj  p2  todo         2  a7   Small
$ ergon ls --sort attention
FTFE  p1  todo         8  a2   Big
qxpj  p2  todo         2  a7   Small
nHcx  p0  todo         -  a?   Unrated
$ ergon ls --sort attention desc
qxpj  p2  todo         2  a7   Small
FTFE  p1  todo         8  a2   Big
nHcx  p0  todo         -  a?   Unrated

# --priority, --attention and --size keep the rows whose value is in a range,
# both ends included; none keeps those without one. Flags combine.
$ ergon ls --attention ..5
FTFE  p1  todo         8  a2   Big
$ ergon ls --attention 7..
qxpj  p2  todo         2  a7   Small
$ ergon ls --attention none
nHcx  p0  todo         -  a?   Unrated
$ ergon ls --priority p0..p1
nHcx  p0  todo         -  a?   Unrated
FTFE  p1  todo         8  a2   Big
$ ergon ls --size 3..
FTFE  p1  todo         8  a2   Big
$ ergon ls --priority ..p1 --attention 2
FTFE  p1  todo         8  a2   Big
$ ergon ls --attention 3..1
error: invalid value '3..1' for '--attention <RANGE>': invalid range "3..1": it starts after it ends

For more information, try '--help'.
[exit 2]
$ ergon ls --attention ..
error: invalid value '..' for '--attention <RANGE>': invalid range "..": give at least one end, as in 2..5, ..3 or 7..

For more information, try '--help'.
[exit 2]
$ ergon ls --priority p9..
error: invalid value 'p9..' for '--priority <RANGE>': invalid priority "p9": expected p0, p1, p2, p3 or p4

For more information, try '--help'.
[exit 2]

# --status keeps tasks with one of the statuses given, effective, so a
# subtask follows its parent's. ls lists only open tasks, so only todo and
# in_progress make sense.
$ ergon edit FTFE --status in_progress
$ ergon ls --status todo
nHcx  p0  todo         -  a?   Unrated
qxpj  p2  todo         2  a7   Small
$ ergon ls --status in-progress,todo --attention ..5
FTFE  p1  in_progress  8  a2   Big
$ ergon ls --status done
error: invalid value 'done' for '--status <STATUS,...>': ergon ls lists only todo and in_progress tasks, never done; see them in ergon tree

For more information, try '--help'.
[exit 2]
