$ ergon init demo "Build a demo\n\nProject-wide context." --no-launch

# References with an explicit project work from anywhere.
$ cd elsewhere
$ ergon edit demo/FTFE --intent "Write the API\nUse GraphQL."
$ ergon --project demo edit qxpj --priority none
