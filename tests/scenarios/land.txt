# `ergon land-hook` is what `choros land` runs before and after each land,
# as a trunk's land hook, so that one command lands the work and moves its
# tasks. Choros gives it the land as JSON; these are the parts ergon reads.

$ fake-choros <<EOF
{"queues": [{"name": "q", "sets": [
  {"n": 3, "repo": "app", "state": "front", "pr": {"number": 6}, "sessions": [], "labels": ["demo/FTFE"]},
  {"n": 4, "repo": "app", "state": "queued", "pr": null, "sessions": [], "labels": ["demo/qxpj"]}
]}]}
EOF
$ cd ws
$ ergon init demo "Build a demo" --no-launch
Created project demo in $ROOT/ws
$ ergon add Backend --priority p1
qxpj
$ ergon add API --parent qxpj
FTFE
$ ergon add Docs --parent qxpj
nHcx
$ ergon add Guide --parent nHcx --status todo
pQyk
$ ergon attach qxpj --session 0b5e0b5e-0000-4000-8000-000000000001
attached session 0b5e0b5e-0000-4000-8000-000000000001 to demo/qxpj; `ergon prompt demo/qxpj` prints its instructions

# A land that names no task carries the session's. Choros is told to label
# the set with them, and the most urgent priority among them. A body that
# was written gets the tasks named under it.
$ ergon land-hook before <<EOF
{"mode": "continue", "labels": [], "priority": null, "title": null, "body": "What changed.", "session": "0b5e0b5e-0000-4000-8000-000000000001"}
EOF
{"labels":["demo/qxpj"],"priority":"p1","body":"What changed.\n\nTasks:\n- demo/qxpj  Backend\n"}
$ ergon land-hook after <<EOF
{"mode": "continue", "labels": ["demo/qxpj"], "priority": "p1", "title": null, "body": "What changed.", "session": "0b5e0b5e-0000-4000-8000-000000000001"}
EOF
$ ergon tree
root  in_progress  p-  Build a demo
  qxpj  in_progress  p1  Backend
    FTFE  in_progress  p1  API
    nHcx  in_progress  p1  Docs
      pQyk  todo         p1  Guide

# A land that carries nothing of ergon's is left alone.
$ ergon land-hook before <<EOF
{"mode": "continue", "labels": ["JIRA-12"], "priority": null, "title": null, "body": null, "session": null}
EOF

# A piece that carries a subtask of the task being worked on finishes it:
# the subtask goes to landing, unless work under it is still open.
$ ergon land-hook before <<EOF
{"mode": "continue", "labels": ["demo/FTFE", "demo/nHcx"], "priority": "p3", "title": "The API (demo/FTFE, demo/nHcx)", "body": "What changed.", "session": null}
EOF
{"labels":["demo/FTFE","demo/nHcx"]}
$ ergon land-hook after <<EOF
{"mode": "continue", "labels": ["demo/FTFE", "demo/nHcx"], "priority": "p3", "title": "The API (demo/FTFE, demo/nHcx)", "body": "What changed.", "session": null}
EOF
demo/FTFE is now landing
demo/nHcx stays in_progress: demo/pQyk under it is still open
$ ergon tree
root  in_progress  p-  Build a demo
  qxpj  in_progress  p1  Backend
    FTFE  landing      p1  API
    nHcx  in_progress  p1  Docs
      pQyk  todo         p1  Guide

# A subtask with a session of its own is being worked on there, so a piece
# of it leaves it as it is.
$ ergon attach nHcx --session 0b5e0b5e-0000-4000-8000-000000000002
attached session 0b5e0b5e-0000-4000-8000-000000000002 to demo/nHcx; `ergon prompt demo/nHcx` prints its instructions
$ ergon land-hook after <<EOF
{"mode": "continue", "labels": ["demo/nHcx"], "priority": null, "title": null, "body": null, "session": "0b5e0b5e-0000-4000-8000-000000000002"}
EOF
$ ergon land-hook before <<EOF
{"mode": "archive", "labels": [], "priority": null, "title": null, "body": null, "session": "0b5e0b5e-0000-4000-8000-000000000002"}
EOF
