# `ergon land-hook` is the land hook of a workspace `ergon work` made for
# tasks: `choros land` runs it before and after each land from there, so
# that one command lands the work and moves the tasks. Choros gives it the
# land as JSON; these are the parts ergon reads.

$ fake-choros <<EOF
{"queues": [{"name": "q", "sets": [
  {"n": 3, "repo": "app", "state": "front", "pr": {"number": 6}, "sessions": [], "labels": ["demo/FTFE"]},
  {"n": 4, "repo": "app", "state": "queued", "pr": null, "sessions": [], "labels": ["demo/qxpj"]}
]}]}
EOF
$ ergon sync
$ROOT: 2 events from choros
$ cd ws
$ ergon init demo "Build a demo" --no-launch
Created project demo in $ROOT/ws
$ ergon add Backend --priority p1
qxpj
$ ergon add API --parent qxpj
FTFE
$ ergon add Docs --parent qxpj
nHcx
$ ergon add Guide --parent nHcx --status todo
pQyk
$ ergon attach qxpj --session 0b5e0b5e-0000-4000-8000-000000000001
attached session 0b5e0b5e-0000-4000-8000-000000000001 to demo/qxpj; `ergon prompt demo/qxpj` prints its instructions

# A land that names no task carries the ones the workspace is bound to.
# Choros is told to label the set with them, and the most urgent priority
# among them. A body that was written gets the tasks named under it.
$ ergon land-hook --task demo/qxpj before <<EOF
{"mode": "continue", "labels": [], "priority": null, "title": null, "body": "What changed."}
EOF
{"labels":["demo/qxpj"],"priority":"p1","body":"What changed.\n\nTasks:\n- demo/qxpj  Backend\n"}
$ ergon land-hook --task demo/qxpj after <<EOF
{"mode": "continue", "labels": ["demo/qxpj"], "priority": "p1", "title": null, "body": "What changed."}
EOF
$ ergon tree
root  in_progress  p-  Build a demo
  qxpj  in_progress  p1  Backend
    FTFE  in_progress  p1  API
    nHcx  in_progress  p1  Docs
      pQyk  todo         p1  Guide

# A piece that carries a subtask of a bound task finishes it: the subtask
# goes to landing, unless work under it is still open. The bound task
# itself goes on, named or not.
$ ergon land-hook --task demo/qxpj before <<EOF
{"mode": "continue", "labels": ["demo/FTFE", "demo/nHcx", "demo/qxpj"], "priority": "p3", "title": "The API (demo/FTFE, demo/nHcx, demo/qxpj)", "body": "What changed."}
EOF
{"labels":["demo/FTFE","demo/nHcx","demo/qxpj"]}
$ ergon land-hook --task demo/qxpj after <<EOF
{"mode": "continue", "labels": ["demo/FTFE", "demo/nHcx", "demo/qxpj"], "priority": "p3", "title": "The API (demo/FTFE, demo/nHcx, demo/qxpj)", "body": "What changed."}
EOF
demo/FTFE is now landing
demo/nHcx stays in_progress: demo/pQyk under it is still open
$ ergon tree

  qxpj  in_progress  p1  Backend
    FTFE  landing      p1  API
    nHcx  in_progress  p1  Docs
      pQyk  todo         p1  Guide

# `choros land archive` says the work is all out and takes the workspace,
# so it carries the bound tasks whatever its labels name, and is refused,
# before anything lands, while a subtask is open.

{"mode": "archive", "labels": ["demo/FTFE"], "priority": null, "title": null, "body": null}
