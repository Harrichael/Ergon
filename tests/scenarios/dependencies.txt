$ ergon dep add qxpj qxpj
error: refused: demo/qxpj can't depend on itself
[exit 1]
$ ergon dep add FTFE qxpj
error: refused: demo/FTFE can't depend on demo/qxpj, its ancestor
[exit 1]
$ ergon dep add qxpj FTFE
error: refused: demo/qxpj can't depend on demo/FTFE, its subtask
[exit 1]
$ ergon dep add qxpj root
error: refused: the root task can't have dependencies or dependents
[exit 1]

# Direct and indirect cycles. A1 inherits A's wait on B, and A can't finish
# before A1, so C depending on A1 also deadlocks.
$ ergon dep add nHcx pQyk
$ ergon dep add pQyk qxpj
error: refused: these tasks would wait on each other: demo/qxpj -> demo/nHcx -> demo/pQyk -> demo/qxpj
[exit 1]
$ ergon dep add pQyk FTFE
error: refused: these tasks would wait on each other: demo/FTFE -> demo/nHcx -> demo/pQyk -> demo/FTFE
[exit 1]
$ ergon dep add FTFE pQyk

# Moving can't put a dependency under its dependent.
$ ergon mv qxpj nHcx
error: refused: demo/qxpj can't depend on demo/nHcx, its ancestor
[exit 1]

$ ergon dep rm qxpj nHcx
$ ergon dep rm qxpj nHcx
error: demo/qxpj doesn't depend on demo/nHcx
[exit 1]
$ ergon tree
root  in_progress  p-  Root
  qxpj  todo         p-  A
    FTFE  todo         p-  A1  blocked
  nHcx  todo         p-  B  blocked
  pQyk  todo         p-  C

# The root can't be a dependent either, and moving a dependency under its
# dependent is refused too.
$ ergon dep add root qxpj
error: refused: the root task can't have dependencies or dependents
[exit 1]
$ ergon mv pQyk FTFE
error: refused: demo/FTFE can't depend on demo/pQyk, its subtask
[exit 1]

# A task can't finish before its subtasks. Here B waits on A, so A1 can't
# wait on B1: A1 -> B1 -> (B's wait) A -> (A's subtask) A1.
$ ergon add P

$ ergon add A --parent 9cDQ

$ ergon add A1 --parent dFR6

$ ergon add B --parent 9cDQ

$ ergon add B1 --parent AENA

$ ergon dep add AENA dFR6
$ ergon dep add qwpT FubR
error: refused: these tasks would wait on each other: demo/dFR6 -> demo/qwpT -> demo/FubR -> demo/dFR6
[exit 1]
$ ergon add C --parent 9cDQ
